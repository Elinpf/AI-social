## 5. 可复算命令与输出（同一环境、只读两矩阵）

**脚本**（置于临时目录运行，不写入仓库）：

```python
import re
BASE = "<repo>/v2/evidence/"
COLS = ["Q1","Q2","Q3","Q3b","Q4","Q5","Q6","Q7","P"]

def parse(path, idx):          # idx = 判据列起点
    rows = {}
    for line in open(BASE+path, encoding="utf-8"):
        if not line.startswith("|"): continue
        cells = [c.strip().replace("*","") for c in line.strip().strip("|").split("|")]
        if len(cells) < idx + len(COLS): continue
        tid = None
        for c in cells[:3]:
            m = re.search(r"TH-[A-Za-z0-9\-]+", c)
            if m: tid = m.group(0); break
        if tid: rows[tid] = cells[idx:idx+len(COLS)]
    return rows

ev = parse("d12c_batch5_eval_researcher_matrix.md", 2)        # | tid | R | Q1...
pl = parse("d12c_batch5_blind_matrix_prompt_linguist.md", 4)  # | # | tid | R | 台词行 | Q1...
P2 = {"TH-P004-Gplace-market-75":"1"}
P3 = {"TH-P004-Gplace-market-75":"1","TH-P007-P006-72":"1",
      "TH-P005-P006-30":"1","TH-P007-P005-4":"1"}

def num(x):
    try: return int(x)
    except Exception: return None

def stats(patch):
    n=a=nc=0
    for t in pl:
        if t not in ev: continue
        for i,c in enumerate(COLS):
            n += 1
            va = pl[t][i]
            if patch and c=="P" and t in patch: va = patch[t]
            x,y = num(va), num(ev[t][i])
            if x is None or y is None: nc += 1; continue
            if x == y: a += 1
    return n, nc, n-nc, a

for tag,p in (("V1",None),("V2",P2),("V3",P3)):
    n,nc,comp,a = stats(p)
    print("%s 名义=%d 不可对=%d 可对=%d 一致=%d 率=%.4f" % (tag,n,nc,comp,a,a/comp))
```
