```python
# -*- coding: utf-8 -*-
"""D9g 步骤1 抽样 v2 —— 按 amendment1（75b0ae90 / 80 行）修订：
  · 分层阈值按【两臂合计】词频：≥6 ⇒ 目标 6 条/词；否则全抽
  · 跨臂：同词每臂按 msg_uid 升序取前 3，不足则全取（amendment 未改此条）
  · 必带字段加：msg_id / 剥离旁白后文本 / UNPAIRED_BRACKET
  · 旁白剥离：全角（）与半角() 都必须【成对】才剥；不成对 ⇒ 不剥 + 标 UNPAIRED_BRACKET
  · 零命中词 11 个单列
确定性：全部按 msg_uid 升序，不人工挑。
"""
import json, re, sys, collections

MSG=("MSG_DM","MSG_PUBLIC","MSG_GROUP")
PAIR = re.compile(r"[（(][^（）()]*[）)]")     # 成对括号（全角/半角混用也按最近配对处理）
ANY_BR = re.compile(r"[（）()]")
sys.path.insert(0,".worktrees/a2-trigger/code")
from life import dialogue_repair as DR
WORDS=list(DR.NEG_WORDS); assert len(WORDS)==18
PER_ARM_K=3
ARMS=[("d9_a1","v2/evidence/run_d9_a1_n16r84_seed42/run.jsonl"),
      ("f6off","v2/evidence/run_d1_f6off_seed42_n16r84/run.jsonl")]

def strip_narration(body):
    """反复去掉成对括号块；若残留未配对括号 ⇒ 返回 (text, True)"""
    prev=None; cur=body
    while prev!=cur:
        prev=cur; cur=PAIR.sub("", cur)
    unpaired = bool(ANY_BR.search(cur))
    return cur.strip(), unpaired

def load(p):
    out=[]
    for line in open(p,encoding="utf-8"):
        line=line.strip()
        if not line: continue
        d=json.loads(line)
        if d.get("type") not in MSG: continue
        pl=d["payload"]; body=pl.get("body") or ""
        hits=[w for w in WORDS if w in body]        # 抽样口径 = 生产口径（全串）
        if not hits: continue
        dlg, unpaired = strip_narration(body)
        out.append({"msg_uid":pl.get("msg_uid"),"msg_id":pl.get("msg_id"),
                    "body":body,"dialogue":dlg,"unpaired":unpaired,"hits":hits,
                    "from":pl.get("from") or pl.get("author"),
                    "participants":pl.get("participants") or [],
                    "thread_id":pl.get("thread_id"),"round":d.get("round"),
                    "type":d.get("type"),"turn":pl.get("turn_index")})
    return out

data={t:load(p) for t,p in ARMS}
freq={t:collections.Counter(w for r in data[t] for w in r["hits"]) for t,_ in ARMS}
total=collections.Counter()
for t in freq: total.update(freq[t])
print("=== 词频（子串、全串口径）===")
for t,_ in ARMS: print(f"  {t}: 命中 {len(data[t])} 条 | {dict(sorted(freq[t].items(), key=lambda kv:-kv[1]))}")
print(f"  合计: {dict(sorted(total.items(), key=lambda kv:-kv[1]))}")
zero=[w for w in WORDS if total.get(w,0)==0]
print(f"  零命中词（{len(zero)} 个）: {zero}")

HIGH=[w for w in WORDS if total.get(w,0)>=6]
print(f"\n=== 分层（合计≥6 ⇒ 目标6条/词；单臂取前{k_PER if False else PER_ARM_K}不足全取）high={HIGH} ===")
picked={}
for w in WORDS:
    for t,_ in ARMS:
        rows=sorted([r for r in data[t] if w in r["hits"]], key=lambda r:r["msg_uid"])
        take = rows[:PER_ARM_K] if w in HIGH else rows
        for r in take: picked.setdefault((t,r["msg_uid"]), dict(r, arm=t))
sel=sorted(picked.values(), key=lambda r:(r["arm"], r["msg_uid"]))
print(f"=== 抽样结果：{len(sel)} 条 ===")
print("  按臂:", dict(collections.Counter(r['arm'] for r in sel)))
print("  按词:", dict(collections.Counter(w for r in sel for w in r['hits'])))
print("  UNPAIRED_BRACKET 条数:", sum(1 for r in sel if r['unpaired']))
# 目标 36 的缺口诊断
print("\n=== 目标 6 条/词 的实际达成 ===")
for w in HIGH:
    n=sum(1 for r in sel if w in r["hits"])
    a=sum(1 for r in data["d9_a1"] if w in r["hits"]); b=sum(1 for r in data["f6off"] if w in r["hits"])
    print(f"  {w}: 取 {n} 条（a1 {a}→{min(a,PER_ARM_K)}，f6off {b}→{min(b,PER_ARM_K)}）{'  ← 不足6' if n<6 else ''}")
json.dump(sel, open("/tmp/d9g_sample_v2.json","w",encoding="utf-8"), ensure_ascii=False, indent=1)
print("\n=== 表格 ===")
print("| # | msg_uid | msg_id | 臂 | 命中词 | body 原文（【旁白】标出） | 剥离旁白后文本 | UNPAIRED | from | participants | thread_id | round |")
print("|---|---|---|---|---|---|---|---|---|---|---|---|")
def mark(b):
    out=b; 
    for m in PAIR.finditer(b): pass
    # 标出所有成对括号块
    return PAIR.sub(lambda m: "【旁白】"+m.group(0)+"【/旁白】", b)
for i,r in enumerate(sel,1):
    esc=lambda s: str(s).replace("|","\\|").replace("\n"," / ")
    print(f"| {i} | `{r['msg_uid']}` | `{r['msg_id']}` | {r['arm']} | {'/'.join(r['hits'])} | {esc(mark(r['body']))} | {esc(r['dialogue'])} | {'YES' if r['unpaired'] else '-'} | {r['from']} | {','.join(r['participants'])} | `{r['thread_id']}` | {r['round']} |")
```
