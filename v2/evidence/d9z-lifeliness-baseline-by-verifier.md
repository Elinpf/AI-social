## 复算命令

```python
# D1 / D2
import json, collections, itertools
rows=[]
for l in open("v2/evidence/d9z-on_n16r56_s42/run.jsonl",encoding="utf-8"):
    e=json.loads(l)
    if not e.get("type","").startswith("MSG"): continue
    pl=e["payload"]
    rows.append(dict(uid=pl["msg_uid"], frm=pl.get("from") or pl.get("author"),
                     day=pl.get("day"), slot=pl.get("slot"),
                     tr=(pl.get("ctx") or {}).get("trigger"),
                     place=pl.get("place"), reply_to=pl.get("reply_to")))
idx={r["uid"]:r for r in rows}
ps=[(idx[r["uid"]], idx[r["reply_to"]]) for r in rows if r["reply_to"] in idx]
cross=sum(1 for a,b in ps if a["place"]!=b["place"])          # 180
# D1 一致率（仅发起）
g=collections.defaultdict(list)
for r in rows:
    if r["tr"] and r["tr"]!="reply": g[(r["frm"],r["slot"])].append((r["day"],r["tr"]))
ag=tot=0
for k,v in g.items():
    for (d1,t1),(d2,t2) in itertools.combinations(v,2):
        if d1==d2: continue
        tot+=1; ag+=(t1==t2)
# ag/tot = 0.6488 ; 随机基线 Σp² = 0.1365（按"仅发起"的 trigger 边际分布）
```
